{%- else -%} {%- comment -%} One statement, as the SQL shortcode requires. The CTE resolves the person's event first, then every occurrence of that event is listed as a week -- rather than listing only the occurrences their group happens to be mapped to, which would silently drop a week if only one occurrence carries the linkage row. {%- endcomment -%} {% sql personid:'{{ CurrentPerson.Id }}' return:'weeks' %} WITH MyEvent AS ( SELECT TOP 1 eio.EventItemId FROM GroupMember gm JOIN EventItemOccurrenceGroupMap m ON m.GroupId = gm.GroupId JOIN EventItemOccurrence eio ON eio.Id = m.EventItemOccurrenceId WHERE gm.PersonId = @personid AND gm.IsArchived = 0 AND gm.GroupMemberStatus = 1 AND eio.EventItemId IN (508, 509, 510, 511) ORDER BY eio.EventItemId ) SELECT ROW_NUMBER() OVER (ORDER BY s.EffectiveStartDate, eio.Id) AS WeekNumber, eio.Id AS OccurrenceId, ei.Name AS EventName, ISNULL(c.Name, '') AS CampusName, ISNULL(eio.Location, '') AS LocationText, s.EffectiveStartDate AS WeekDate, ISNULL(eio.ContactEmail, '') AS ContactEmail, ISNULL(p.NickName + ' ' + p.LastName, '') AS ContactName FROM EventItemOccurrence eio JOIN MyEvent me ON me.EventItemId = eio.EventItemId JOIN EventItem ei ON ei.Id = eio.EventItemId LEFT JOIN Campus c ON c.Id = eio.CampusId LEFT JOIN Schedule s ON s.Id = eio.ScheduleId LEFT JOIN PersonAlias pa ON pa.Id = eio.ContactPersonAliasId LEFT JOIN Person p ON p.Id = pa.PersonId ORDER BY s.EffectiveStartDate, eio.Id {% endsql %} {%- if weeks == empty -%}
